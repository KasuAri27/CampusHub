```mermaid
classDiagram

%% 关系定义
User "1" --> "*" Task : 发布 (requester)
User "1" --> "*" Order : 接单 (provider)
Category "1" <-- "*" Task : 属于
Task "1" *-- "0..*" Order : 包含
Order "1" *-- "0..1" Review : 包含
User "1" <-- "*" Notification : 接收
User "1" --> "*" Complaint : 发起 (complainantId)
Order "1" <-- "*" Complaint : 关联

%% 枚举类型
class Role {
    <<enumeration>>
    requester
    provider
    admin
}

class TaskStatus {
    <<enumeration>>
    pending_review
    published
    in_progress
    completed
    cancelled
}

class OrderStatus {
    <<enumeration>>
    accepted
    in_progress
    completed
    cancelled
}

%% 核心实体类
class User {
    -Long id
    -String studentId
    -String name
    -String nickname
    -String email
    -String phone
    -String password
    -Role role
    -Integer creditScore
    -String avatar
    -Integer status
    -LocalDateTime createTime
    -LocalDateTime updateTime
}

class Category {
    -Integer id
    -String name
    -String icon
    -Boolean hasReview
    -Integer sortOrder
    -LocalDateTime createdAt
}

class Task {
    <<abstract>>
    -Long id
    -User requester
    -Category category
    -String title
    -String description
    -BigDecimal reward
    -TaskStatus status
    -String campus
    -String location
    -LocalDateTime deadline
    -int viewCount
    -LocalDateTime publishedAt
    -LocalDateTime createdAt
    -LocalDateTime updatedAt
}

%% Task 子类
class ExpressTask {
    -String pickupLocation
    -String deliveryLocation
    -String pickupCode
    -BigDecimal expressReward
}

class TutoringTask {
    -String subject
    -BigDecimal pricePerHour
    -String tutoringLocation
}

class SecondHandTask {
    -String tradeMode
    -BigDecimal itemPrice
    -String requesterPaymentCode
}

class TeamingTask {
    -int maxMembers
    -Long chatGroupId
}

class ConsultingTask {
    -String consultingMode
}

Task <|-- ExpressTask
Task <|-- TutoringTask
Task <|-- SecondHandTask
Task <|-- TeamingTask
Task <|-- ConsultingTask

class Order {
    -Long id
    -Task task
    -User requester
    -User provider
    -OrderStatus status
    -LocalDateTime acceptedAt
    -LocalDateTime completedAt
    -LocalDateTime cancelledAt
    -String paymentCode
    -LocalDateTime createdAt
    -LocalDateTime updatedAt
}

class Review {
    -Long id
    -Order order
    -Integer score
    -String comment
    -LocalDateTime createdAt
}

class Notification {
    -Long id
    -Long receiverId
    -String title
    -String content
    -String type
    -Boolean isRead
    -LocalDateTime createTime
    -LocalDateTime readTime
}

class Complaint {
    -Long id
    -Long complainantId
    -Long orderId
    -Long targetId
    -String reason
    -String description
    -String imageUrl
    -LocalDateTime createTime
}
